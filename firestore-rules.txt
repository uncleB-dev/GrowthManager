rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // 1. Members: 본인 정보만 수정 가능, 조회는 팀장도 가능
    match /members/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
    
    // 2. Daily Logs: 본인 기록만 수정 가능, 수락된 팀장만 조회 가능
    match /daily_logs/{logId} {
      allow write: if request.auth != null && request.auth.uid == resource.data.uid || !exists(resource);
      allow read: if request.auth != null && (
        request.auth.uid == resource.data.uid || 
        get(/databases/$(database)/documents/connections/$(request.auth.token.email + "_" + resource.data.uid)).data.status == "accepted"
      );
    }
    
    // 3. Connections: 관련 당사자만 접근 가능
    match /connections/{connId} {
      allow read, write: if request.auth != null && (
        request.auth.uid == resource.data.memberUid || 
        request.auth.token.email == resource.data.leaderEmail
      );
    }
  }
}
